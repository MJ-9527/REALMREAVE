# v0.8 计划：一键出售 · 装备职业契合 · 合成品质选择 · 地图差异化

## Summary

四项需求，全部落在纯前端原型 `d:\github\game\prototype` 内：

1. **一键出售**：商店「出售物资」区增加「一键出售」按钮，把当前所有**非锁定**材料一次性按 7 折换成物资点。
2. **装备职业契合**：装备实例新增 `cls`（职业）字段。分近战组（重装 / 近战 / 骑兵）与远程组（远程 / 法术）：同职业 = 100% 性能；同组异职业 = 60%（例：轻骑 + 铁剑）；跨组 = **无法装备**。护甲与饰品不带职业，始终 100%。
3. **合成品质选择**：配方行右侧并排显示每个可用品质的合成按钮（含直接合成与「补齐后合成」两种，按钮上带花费），点哪个品质就消耗哪个品质的主材，便于保留高级材料。
4. **地图差异化**：为森林 / 冰原 / 平原 / 沙漠 / 火山各配一套独立的地图结构参数（环半径、节点密度、压扁率、怪物与物资配比、额外长边），同一局内强制三张图来自三个不同地形。

```js
/* ---------------- 装备职业契合 ---------------- */
const APTI_GROUP = {
  '重装': 'melee', '近战': 'melee', '骑兵': 'melee',
  '远程': 'ranged', '法术': 'ranged'
};
const APTI_PERFECT = 1;
const APTI_PARTIAL = 0.6;

/** 装备对该职业的契合度：perfect / partial / none */
function aptitudeOf(cls, eq) {
  if (!eq || !eq.cls) return 'perfect';      // 护甲 / 饰品 / 无职业武器
  if (eq.cls === cls) return 'perfect';
  if (APTI_GROUP[eq.cls] && APTI_GROUP[eq.cls] === APTI_GROUP[cls]) return 'partial';
  return 'none';
}
function aptitudeMod(cls, eq) {
  const a = aptitudeOf(cls, eq);
  return a === 'perfect' ? APTI_PERFECT : a === 'partial' ? APTI_PARTIAL : 0;
}
```

### 2. units.js —— 装备带职业、加成按契合打折

```js
function makeEquip(slot, qKey, name, cls) {
  let c = cls || null;
  if (slot === 'weapon' && !c) c = CLASSES[randInt(0, CLASSES.length - 1)];
  return {
    uid: 'e' + (++equipSeq),
    slot,
    quality: qKey,
    cls: c,                       // 武器：近战/远程/法术/骑兵/重装；护甲/饰品：null
    name: name || defaultEquipName(slot, qKey, c)
  };
}

function defaultEquipName(slot, qKey, cls) {
  const q = QMAP[qKey];
  if (slot === 'weapon') return q.name + (WEAPON_NAMES[cls] || '剑');
  if (slot === 'armor') return q.name + ARMOR_NAMES[randInt(0, 2)];
  return q.name + TRINKET_NAMES[randInt(0, 2)];
}

/** 装备加成；传 ownerCls 时按契合度打折 */
function equipBonus(equips, ownerCls) {
  let atk = 0, hp = 0, def = 0;
  for (const e of equips) {
    if (!e) continue;
    const m = QMAP[e.quality].mult * (ownerCls ? aptitudeMod(ownerCls, e) : 1);
    if (e.slot === 'weapon') atk += 4 * m;
    else if (e.slot === 'armor') { hp += 12 * m; def += 2 * m; }
    else { atk += 1.5 * m; hp += 6 * m; def += 1 * m; }
  }
  return { atk: Math.round(atk), hp: Math.round(hp), def: Math.round(def) };
}
```

`heroToBattleUnit(player)` 改为 `equipBonus(player.equips, player.cls)`。

不传 `ownerCls` 时行为与现在一致（用于背包里展示原始数值）。

#### 4.3 合成品质选择

新增（放在 `craftCheck` 附近）：

```js
/** 该配方所有可执行的品质选项（高品质在前） */
function craftOptions(p, r) {
  const auxNeed = {};
  for (const a of r.aux) auxNeed[a] = (auxNeed[a] || 0) + 1;
  const out = [];
  for (let i = QUALITIES.length - 1; i >= 0; i--) {
    const q = QUALITIES[i].key;
    const hasMain = p.materials.some(m => m.n === r.main && m.q === q);
    const miss = [];
    let blocked = false;
    for (const name in auxNeed) {
      const lack = auxNeed[name] - matCount(p, name);
      if (lack <= 0) continue;
      if (!BASE_MATERIALS.includes(name)) { blocked = true; break; }
      miss.push({ name, lack, q: matSupplyTop(G.round) });
    }
    if (blocked) continue;
    if (!hasMain) {
      if (!BASE_MATERIALS.includes(r.main)) continue;
      if (!matSupplyTiers(G.round).includes(q)) continue;
      miss.push({ name: r.main, lack: 1, q });
    }
    const cost = miss.reduce((s, m) => s + m.lack * priceFor(p, MAT_PRICE[m.q]), 0);
    out.push({ q, miss, cost, can: cost <= p.credit });
  }
  return out;
}

/** 取指定材料里品质最低的一件（辅材优先吃低品质，保住高级材料） */
function takeLowestQuality(p, name) {
  let idx = -1;
  for (let i = 0; i < p.materials.length; i++) {
    const m = p.materials[i];
    if (m.n !== name) continue;
    if (idx < 0 || QINDEX[m.q] < QINDEX[p.materials[idx].q]) idx = i;
  }
  if (idx < 0) return null;
  return p.materials.splice(idx, 1)[0];
}

/** 按指定品质合成（必要时先按商店价补齐） */
function craftAt(id, qKey) {
  const me = G.me;
  const r = RECIPES.find(x => x.id === id);
  if (!me || !r) return;
  const opt = craftOptions(me, r).find(o => o.q === qKey);
  if (!opt) return;
  if (opt.cost > 0 && !opt.can) {
    log(`物资点不足：补齐需 ${opt.cost}，当前 ${budgetOf(me)}，差 ${opt.cost - budgetOf(me)}`, '');
    renderShop();
    return;
  }
  if (opt.cost > 0) {
    payFrom(me, opt.cost);
    for (const m of opt.miss) {
      for (let i = 0; i < m.lack; i++) me.materials.push({ n: m.name, q: m.q });
    }
  }
  takeMaterial(me, r.main, qKey);
  for (const a of r.aux) takeLowestQuality(me, a);
  const eq = makeEquip(r.slot, qKey, r.name, r.cls);
  me.equipBag.push(eq);
  log(`合成 <span class="q-${qKey}">${esc(eq.name)}</span>${opt.cost > 0 ? `（补齐 -${opt.cost} 物资点）` : ''}，已放入装备背包`, 'hl');
  renderShop();
  renderSide();
}
```

## Assumptions & Decisions

- **契合规则**（用户已确认「先按近战和远程分，后续补充具体说明」）：同职业 100%，同组异职业 60%，跨组禁止。护甲 / 饰品不带职业，始终 100%。用户原例「轻骑 + 铁剑」→ 骑兵与近战同属近战组 → 60%，与本实现一致。
- **掉落 / 商店武器**沿用随机职业（`makeEquip` 自动分配），因此会出现「用不了的武器」，可按 5 折在背包卖出；合成武器固定为配方职业。
- **一键出售不做二次确认**（用户要求「直接出售」）；锁定的金及以上品质与当前可合成配方的主材天然被排除，避免误卖。
- **合成品质**（用户已确认「每个品质一个按钮」）：只列出真正可执行的品质——要么已有该品质主材，要么该品质处于本轮材料补给档位且主材是基础材料。补齐的辅材仍按当轮最高档计价，补齐的主材按所选品质计价。
- **保留高级材料**：辅材改为优先消耗最低品质；主材由玩家选择的品质决定，未被选中的高品质主材不会被消耗（当前合成锁仍只锁最高品质主材实例，属既有行为，不改）。
- **地图**（用户已确认「每地形一套结构参数」）：程序化生成 + 每局三图地形不重复；不引入固定手工坐标。
